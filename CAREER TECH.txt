## 🗄️ Struktur Tabel MySQL
```sql
CREATE TABLE careers (
  id INT AUTO_INCREMENT PRIMARY KEY,
  title VARCHAR(150),
  description TEXT,
  requirements TEXT,
  email VARCHAR(150)
);
```

- `title` → Nama posisi (misalnya "Textile Production Operator").
- `description` → Job description (pakai format teks panjang).
- `requirements` → Requirements & Qualifications.
- `email` → Alamat email HR untuk apply.

---

## 📌 Model (`Career_model.php`)
```php
<?php
class Career_model extends CI_Model {

    public function get_all() {
        return $this->db->get('careers')->result();
    }

    public function insert($data) {
        return $this->db->insert('careers', $data);
    }

    public function get_by_id($id) {
        return $this->db->get_where('careers', ['id' => $id])->row();
    }

    public function update($id, $data) {
        $this->db->where('id', $id);
        return $this->db->update('careers', $data);
    }

    public function delete($id) {
        return $this->db->delete('careers', ['id' => $id]);
    }
}
```

---

## 📌 Controller (`Career.php`)
```php
<?php
class Career extends CI_Controller {

    public function __construct() {
        parent::__construct();
        $this->load->model('Career_model');
    }

    public function index() {
        $data['careers'] = $this->Career_model->get_all();
        $this->load->view('career_list', $data);
    }
}
```

## 📌 Controller (`Career_admin.php`)
```php
<?php
class Career_admin extends CI_Controller {

    public function __construct() {
        parent::__construct();
        $this->load->model('Career_model');
    }

    public function index() {
        $data['careers'] = $this->Career_model->get_all();
        $this->load->view('career_admin/list', $data);
    }

    public function create() {
        if ($this->input->post()) {
            $data = [
                'title' => $this->input->post('title'),
                'description' => $this->input->post('description'),
                'requirements' => $this->input->post('requirements'),
                'email' => $this->input->post('email')
            ];
            $this->Career_model->insert($data);
            redirect('career_admin');
        }
        $this->load->view('career_admin/create');
    }

    public function edit($id) {
        $data['career'] = $this->Career_model->get_by_id($id);
        if ($this->input->post()) {
            $update = [
                'title' => $this->input->post('title'),
                'description' => $this->input->post('description'),
                'requirements' => $this->input->post('requirements'),
                'email' => $this->input->post('email')
            ];
            $this->Career_model->update($id, $update);
            redirect('career_admin');
        }
        $this->load->view('career_admin/edit', $data);
    }

    public function delete($id) {
        $this->Career_model->delete($id);
        redirect('career_admin');
    }
}
```
